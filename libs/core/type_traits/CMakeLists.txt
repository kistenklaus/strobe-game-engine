

set(STROBE_CORE_TYPE_TRAITS_LIB
  ${STROBE_CORE_LIB}-type-traits CACHE INTERNAL "")


add_library(${STROBE_CORE_TYPE_TRAITS_LIB} INTERFACE)

target_include_directories(${STROBE_CORE_TYPE_TRAITS_LIB}
  INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/include)

set(COMPILER_ID "${CMAKE_CXX_COMPILER_ID}")
if (COMPILER_ID STREQUAL "GNU" OR COMPILER_ID STREQUAL "Clang")
  target_link_libraries(${STROBE_CORE_TYPE_TRAITS_LIB} INTERFACE atomic)
endif()
